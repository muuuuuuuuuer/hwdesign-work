//        bit 9  - interrupt (Read)
//        others - reserved
// 0x04 : Global Interrupt Enable Register
//        bit 0  - Global Interrupt Enable (Read/Write)
//        others - reserved
// 0x08 : IP Interrupt Enable Register (Read/Write)
//        bit 0 - enable ap_done interrupt (Read/Write)
//        bit 1 - enable ap_ready interrupt (Read/Write)
//        others - reserved

//        bit 0 - ap_done (Read/TOW)
//        bit 1 - ap_ready (Read/TOW)
//        others - reserved
// 0x10 : Data signal of halted
//        bit 31~0 - halted[31:0] (Read)
// 0x14 : Control signal of halted
//        bit 0  - halted_ap_vld (Read/COR)
//        others - reserved
// 0x20 : Data signal of error
//        bit 31~0 - error[31:0] (Read)
// 0x24 : Control signal of error
//        bit 0  - error_ap_vld (Read/COR)
//        others - reserved
// 0x30 : Data signal of error_tx_id
//        bit 31~0 - error_tx_id[31:0] (Read)
// 0x34 : Control signal of error_tx_id
//        bit 0  - error_tx_id_ap_vld (Read/COR)
//        others - reserved
// (SC = Self Clear, COR = Clear on Read, TOW = Toggle on Write, COH = Clear on Handshake)
